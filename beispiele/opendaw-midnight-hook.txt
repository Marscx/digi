// openDAW Script: 8-Takt Hip-Hop Beat in A-Moll, 88 BPM
// Drums (Playfield), 808-Bass, Pad, Hookline
// openDAW > Script Editor > einfügen > Run

const BARS = 8
const S = PPQN.SemiQuaver // 1/16
const BAR = PPQN.Bar

const project = openDAW.newProject("Midnight Hook")
project.bpm = 88

// kleine Humanisierung (deterministisch)
let seed = 7
const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647
const human = (amount: number) => Math.round((rnd() - 0.5) * amount)

// ---------- Samples ----------
const samples = await openDAW.listSamples()
// Wunsch-Samples aus der openDAW-Library, sonst irgendein kurzer One-Shot (keine Loops)
const oneShots = samples.filter(sample => sample.duration < 1.5 && !sample.name.toLowerCase().includes("loop"))
const find = (...needles: string[]) => {
    for (const needle of needles) {
        const hit = oneShots.find(sample => sample.name.toLowerCase().includes(needle.toLowerCase()))
        if (hit) {return hit}
    }
    return undefined
}
// Fallback: Sounds synthetisieren, falls die Library etwas nicht hat
const synthDrum = async (name: string, seconds: number, fn: (t: number) => number) => {
    const frames = Math.floor(sampleRate * seconds)
    const audio = AudioData.create(sampleRate, frames, 1)
    const data = audio.frames[0]
    for (let i = 0; i < frames; i++) {data[i] = fn(i / sampleRate)}
    return openDAW.addSample(audio, name)
}
let kickPhase = 0
const kick = find("Dusty Kick Crunch", "ReelRtm Deep Kick", "Kick") ?? await synthDrum("Synth Kick", 0.6, t => {
    kickPhase += (45 + 120 * Math.exp(-t * 30)) / sampleRate
    return Math.sin(kickPhase * Math.PI * 2) * Math.exp(-t * 6) * 0.9
})
const snare = find("Dusty Snare Lug", "LoFi2 Snare Chunk", "Snare") ?? await synthDrum("Synth Snare", 0.3, t =>
    (Math.random() * 2 - 1) * Math.exp(-t * 18) * 0.6)
const clap = find("909 Clap", "TR-707 Clap", "Clap") ?? await synthDrum("Synth Clap", 0.35, t =>
    (Math.random() * 2 - 1) * Math.exp(-t * 14) * 0.5)
const hat = find("LoFi2 CHat Drip", "909 Closed Hat", "Hihat Closed", "CHat") ?? await synthDrum("Synth Hat", 0.08, t =>
    (Math.random() * 2 - 1) * Math.exp(-t * 60) * 0.4)
const openHat = find("Dusty Hat Open Rattle", "909 Open Hat", "Hihat Open", "OHat") ?? await synthDrum("Synth Open Hat", 0.4, t =>
    (Math.random() * 2 - 1) * Math.exp(-t * 9) * 0.3)

// ---------- Mix-Busse ----------
const reverb = project.addAuxUnit({label: "Reverb"})
reverb.addAudioEffect("Reverb", {decay: 0.75, damp: 0.6, preDelay: 0.03, wet: 0, dry: -Infinity})
const delay = project.addAuxUnit({label: "Delay"})
delay.addAudioEffect("Delay", {delay: 13, feedback: 0.35, filter: 0.3, wet: 0, dry: -Infinity})

// ---------- Drums ----------
const drums = project.addInstrumentUnit("Playfield", {label: "Drums", volume: -4})
drums.instrument.addSample(kick, {note: 36})
drums.instrument.addSample(snare, {note: 38})
drums.instrument.addSample(clap, {note: 39, volume: -4})
drums.instrument.addSample(hat, {note: 42, exclude: true, release: 0.05})
drums.instrument.addSample(openHat, {note: 46, exclude: true})
drums.addAudioEffect("Compressor", {threshold: -14, ratio: 4, attack: 5, release: 80})
drums.addSend(reverb, {amount: -24})

const drumRegion = drums.noteTracks[0].addRegion({duration: BAR * BARS, label: "Groove"})
for (let bar = 0; bar < BARS; bar++) {
    const o = bar * BAR
    const fill = bar % 4 === 3
    // Kick: Boom-bap-Trap Hybrid
    const kicks = bar % 2 === 0 ? [0, 7, 10] : [0, 3, 10, 14]
    kicks.forEach(step => drumRegion.addEvent({position: o + step * S, duration: S, pitch: 36, velocity: step === 0 ? 1 : 0.85}))
    // Clap/Snare auf 2 und 4, leicht "laid back"
    ;[4, 12].forEach(step => {
        drumRegion.addEvent({position: o + step * S + 18, duration: S, pitch: 38, velocity: 0.95})
        drumRegion.addEvent({position: o + step * S + 24, duration: S, pitch: 39, velocity: 0.8})
    })
    if (fill) {drumRegion.addEvent({position: o + 15 * S, duration: S, pitch: 38, velocity: 0.45})}
    // Hats: 16tel mit Ghost-Notes und Trap-Rolls
    for (let step = 0; step < 16; step++) {
        if (step === 6 && bar % 2 === 1) {continue} // Platz fürs Open-Hat
        const accent = step % 4 === 0 ? 0.85 : step % 2 === 0 ? 0.6 : 0.35
        const roll = (fill && (step === 13 || step === 14)) || (!fill && bar % 2 === 1 && step === 11)
        drumRegion.addEvent({
            position: Math.max(0, o + step * S + (step % 2 === 1 ? 14 : 0) + human(10)),
            duration: S,
            pitch: 42,
            velocity: Math.min(1, accent + human(10) / 100),
            playCount: roll ? 3 : 1
        })
    }
    if (bar % 2 === 1) {drumRegion.addEvent({position: o + 6 * S, duration: S * 2, pitch: 46, velocity: 0.55})}
}

// ---------- Harmonien: Am9 | Fmaj9 | Dm9 | E7(b9), je 2 Takte ----------
const chords = [
    {root: 33, notes: [57, 60, 64, 67, 71]}, // Am9
    {root: 29, notes: [53, 57, 60, 64, 67]}, // Fmaj9
    {root: 38, notes: [50, 53, 57, 60, 64]}, // Dm9
    {root: 28, notes: [52, 56, 59, 62, 65]}  // E7(b9)
]

// ---------- Pad ----------
const pad = project.addInstrumentUnit("Vaporisateur", {label: "Pad", volume: -16, panning: 0}, {
    cutoff: 1600, resonance: 0.3, filterOrder: 2, filterEnvelope: 0.15,
    attack: 0.8, decay: 1.2, sustain: 0.8, release: 1.6,
    voicingMode: VoicingMode.Polyphonic, unisonCount: 5, unisonDetune: 18, unisonStereo: 1
})
pad.instrument.oscillators[0].waveform = ClassicWaveform.saw
pad.instrument.oscillators[1].waveform = ClassicWaveform.triangle
pad.instrument.oscillators[1].volume = -10
pad.instrument.oscillators[1].octave = 1
pad.addSend(reverb, {amount: -6})
const padRegion = pad.noteTracks[0].addRegion({duration: BAR * BARS, label: "Chords"})
chords.forEach((chord, index) => chord.notes.forEach((pitch, voice) => padRegion.addEvent({
    position: index * BAR * 2 + voice * 6, // leicht "gestrummt"
    duration: BAR * 2 - 60,
    pitch,
    velocity: 0.55
})))

// ---------- 808 Bass ----------
const bass = project.addInstrumentUnit("Vaporisateur", {label: "808 Bass", volume: -6}, {
    cutoff: 900, resonance: 0.2, filterOrder: 2,
    attack: 0.003, decay: 0.8, sustain: 0.55, release: 0.12,
    voicingMode: VoicingMode.Monophonic, glideTime: 0.12
})
bass.instrument.oscillators[0].waveform = ClassicWaveform.sine
bass.instrument.oscillators[0].volume = 0
bass.instrument.oscillators[1].waveform = ClassicWaveform.triangle
bass.instrument.oscillators[1].volume = -16 // Oberton, damit man's auch auf Laptop-Speakern hört
bass.instrument.oscillators[1].octave = 1
const bassRegion = bass.noteTracks[0].addRegion({duration: BAR * BARS, label: "808"})
// [Step (16tel innerhalb 2 Takte), Länge in 16teln, Intervall zum Grundton]
const bassPattern: [number, number, number][] = [
    [0, 6, 0], [7, 2, 0], [10, 5, 0],
    [16, 3, 0], [19, 2, 12], [22, 4, 0], [26, 2, 7], [28, 3, 0]
]
chords.forEach((chord, index) => bassPattern.forEach(([step, length, interval]) => bassRegion.addEvent({
    position: index * BAR * 2 + step * S,
    duration: length * S - 20,
    pitch: chord.root + interval,
    velocity: step % 16 === 0 ? 1 : 0.85
})))

// ---------- Hookline ----------
const hook = project.addInstrumentUnit("Vaporisateur", {label: "Hook", volume: -11, panning: 0.1}, {
    cutoff: 2600, resonance: 0.5, filterOrder: 2, filterEnvelope: 0.45,
    attack: 0.004, decay: 0.28, sustain: 0.25, release: 0.35,
    voicingMode: VoicingMode.Monophonic, glideTime: 0.04, unisonCount: 3, unisonDetune: 12
})
hook.instrument.oscillators[0].waveform = ClassicWaveform.square
hook.instrument.oscillators[1].waveform = ClassicWaveform.saw
hook.instrument.oscillators[1].volume = -12
hook.instrument.oscillators[1].tune = 7
hook.addSend(delay, {amount: -10})
hook.addSend(reverb, {amount: -14})
// [Step, Länge, Pitch] – Frage (Takt 1-2), Antwort (3-4), Frage (5-6), Auflösung (7-8)
const call: [number, number, number][] = [
    [0, 2, 76], [3, 1, 74], [4, 2, 72], [6, 3, 69], [10, 2, 72], [12, 4, 74],
    [16, 2, 76], [19, 1, 79], [20, 2, 76], [22, 2, 74], [24, 6, 72]
]
const answer: [number, number, number][] = [
    [0, 2, 72], [3, 1, 74], [4, 2, 76], [6, 3, 79], [10, 2, 76], [12, 4, 74],
    [16, 2, 72], [19, 1, 69], [20, 2, 72], [22, 2, 74], [24, 6, 69]
]
const resolve: [number, number, number][] = [
    [0, 2, 76], [3, 1, 74], [4, 2, 71], [6, 3, 68], [10, 2, 71], [12, 4, 74],
    [16, 2, 76], [19, 1, 77], [20, 2, 76], [22, 2, 74], [24, 4, 71], [28, 4, 68]
]
const hookRegion = hook.noteTracks[0].addRegion({duration: BAR * BARS, label: "Hook"})
;[call, answer, call, resolve].forEach((phrase, index) => phrase.forEach(([step, length, pitch]) => hookRegion.addEvent({
    position: Math.max(0, index * BAR * 2 + step * S + human(12)),
    duration: length * S - 30,
    pitch,
    velocity: 0.7 + (step % 4 === 0 ? 0.2 : 0) + human(8) / 100
})))

// ---------- Transport ----------
project.loop.enabled = true
project.loop.from = 0
project.loop.to = BAR * BARS

project.openInStudio()
